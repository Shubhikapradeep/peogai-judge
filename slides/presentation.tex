\begin{frame}{2. Problem Statement \& Motivation}
  \begin{columns}
    \begin{column}{0.5\textwidth}
      \textbf{The Promise:}
      \begin{itemize}
        \item Continuous Integration (CI/CD) pipelines require scalable, automated code auditing.
        \item LLM-as-a-Judge offers zero-shot review of syntax, semantics, and correctness.
      \end{itemize}
    \end{column}
    \begin{column}{0.5\textwidth}
      \textbf{The Critical Question \& Vulnerability:}
      \begin{itemize}
        \item Can open-source models ($\le 7$B) running locally on consumer hardware reliably judge code?
        \item Does the order of answer presentation dictate the verdict?
        \item Flawed code could be merged if the evaluator suffers from presentation-order bias.
      \end{itemize}
    \end{column}
  \end{columns}
\end{frame}

% Slide 3
\begin{frame}{3. Literature Summary \& Foundations}
  \begin{itemize}
    \item \textbf{MT-Bench \& Chatbot Arena (Zheng et al., 2023):} Formalized LLM-as-a-judge for natural dialogue; identified position, verbosity, and self-enhancement biases in proprietary models (GPT-4).
    \item \textbf{CodeJudgeBench (Chen et al., 2024):} Benchmarked code evaluation with objective unit-test ground truth.
    \item \textbf{"Bias in the Loop" (Stureborg et al., 2024):} Analyzed length bias in 70B+ models; left edge-scale coder models unexamined.
  \end{itemize}
\end{frame}

% Slide 5
\begin{frame}{5. Research Questions}
  \begin{itemize}
    \item \textbf{RQ1 (Accuracy):} How accurately can small open-source models (3B and 7B) perform pairwise and pointwise code evaluation?
    \item \textbf{RQ2 (Position Bias):} To what extent does presentation order dictate verdicts, and what is the resulting consistency gap?
    \item \textbf{RQ3 (False Acceptance):} In pointwise auditing, what fraction of incorrect solutions are erroneously approved (FAR)?
    \item \textbf{RQ4 (Mitigation Bounds):} Can debiasing prompts or majority voting resolve bias, or is architectural order-averaging required?
  \end{itemize}
\end{frame}

% Slide 6
\begin{frame}{6. Scope: What We Did and Did Not Do}
  \begin{columns}
    \begin{column}{0.5\textwidth}
      \textbf{What We Did:}
      \begin{itemize}
        \item 150 balanced code pairs (300 programs): 50 Easy / 50 Medium / 50 Hard.
        \item Dual-order permutation for all pairs (2,400 calls total).
        \item Pairwise, Pointwise, Verbosity, and Fix evaluations.
      \end{itemize}
    \end{column}
    \begin{column}{0.5\textwidth}
      \textbf{Deliberate Boundaries (Why):}
      \begin{itemize}
        \item No closed proprietary APIs: focus on local, edge-scale models.
        \item No self-enhancement test: models did not judge their own code.
        \item No subjective style/readability metrics: confined to compiler test correctness.
      \end{itemize}
    \end{column}
  \end{columns}
\end{frame}

  \begin{itemize}
    \item \textbf{Size:} 150 pairs (300 solutions), stratified evenly (50 Easy, 50 Medium, 50 Hard).
    \item \textbf{Platforms:} 78 LeetCode, 71 AtCoder, 1 Codeforces.
    \item \textbf{Generators:} Claude 3.7 Sonnet, Gemini 2.5 Flash, Claude 4, Qwen 235B.

\begin{frame}{9. Models \& Experimental Setup}
  \begin{itemize}
    \item \textbf{Models:} Qwen2.5-Coder-3B-Instruct (fp16) and Qwen2.5-Coder-7B-Instruct-AWQ (4-bit).
    \item \textbf{Hardware:} NVIDIA Tesla T4 (Colab) for batch runs; Apple MacBook Pro M3 Pro (Ollama) for demo.
    \item \textbf{Decoding:} Greedy ($T=0$), max 16 tokens; Vote@3 uses $T=0.7$, top-$p=0.95$.
    \item \textbf{Scale:} 2,400 total judge calls, 1.27 GPU-hours, 0.0\% parse failures.
  \end{itemize}
\end{frame}

% Slide 10
\begin{frame}{10. Experiments Overview}
  \begin{itemize}
    \item \textbf{Exp 1 (Pairwise):} Both presentation orders $(1,2)$ and $(2,1)$; measures accuracy gap, consistency, and \% A verdicts.
    \item \textbf{Exp 2 (Verbosity):} Padding incorrect code with docstrings/comments claiming correctness (AST logic preserved).
    \item \textbf{Exp 3 (Pointwise):} Auditing single code snippets; computes False Acceptance Rate (FAR) and False Rejection Rate (FRR).
    \item \textbf{Exp 4 (Fixes):} Dual-order averaging (4a), debias prompt (4b), and majority voting of 3 (4c).
  \end{itemize}
\end{frame}

% Slide 11
\begin{frame}{11. Results: Pairwise \& Pointwise Auditing}
  \begin{columns}
    \begin{column}{0.5\textwidth}
      \textbf{Exp 1: Pairwise Position Bias}
      \begin{itemize}
        \item 3B Correct=A: \textbf{98.7\%} vs Correct=B: \textcolor{red}{\textbf{6.7\%}} (\textbf{+92.0 pp gap})
        \item 3B Consistency across orders: \textcolor{red}{\textbf{8.0\%}}
        \item 3B \% A verdicts: \textbf{96.0\%} (138 AA flips of 150)
        \item 7B gap: \textbf{+83.3 pp}; consistency: \textbf{15.3\%}
      \end{itemize}
    \end{column}
    \begin{column}{0.5\textwidth}
      \textbf{Exp 3: Pointwise Error Rates}
      \begin{itemize}
        \item 3B False Acceptance (FAR): \textcolor{red}{\textbf{76.0\%}}
        \item 7B False Acceptance (FAR): \textcolor{red}{\textbf{72.0\%}}
        \item Labeled correct: \textbf{79.0\%} (3B) and \textbf{78.3\%} (7B)
        \item \emph{Takeaway:} Leniency bias allows 3 out of 4 buggy programs to pass review!
      \end{itemize}
    \end{column}
  \end{columns}
\end{frame}

% Slide 13
\begin{frame}{13. Key Findings \& Limitations}
  \begin{columns}
    \begin{column}{0.55\textwidth}
      \textbf{Five Main Findings:}
      \begin{enumerate}
        \item \textbf{Position Bias:} 3B voted A in 96\% of calls; rare B votes carry true signal (83.3\% right, $p=0.039$).
        \item \textbf{Pointwise Leniency:} 3B accepts 76\% of buggy code (FAR), approving ~79\% of answers.
        \item \textbf{Scale (3B vs 7B):} 7B is less biased ($p=0.011$) but no more accurate ($p=0.405$).
        \item \textbf{Fixes Fail:} Debias prompt worsened 3B bias ($p=0.016$); only order-averaging ensures safety.
        \item \textbf{Verbosity (Exp 2):} Comments only amplify the default to position A ($p=0.004$).
      \end{enumerate}
    \end{column}
    \begin{column}{0.5\textwidth}
      \textbf{Limitations:}
      \begin{itemize}
        \item Difficulty label noise: 82\% conflict across splits in CodeJudgeBench.
        \item Model scale limited to $\le 7$B parameters.
        \item Single-language (Python) focus.
      \end{itemize}
    \end{column}
  \end{columns}
\end{frame}
